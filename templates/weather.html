Search for a city to see its forecast.

const searchButton = document.getElementById("searchWeather"); const cityInput = document.getElementById("cityInput"); searchButton.addEventListener("click", getWeather); cityInput.addEventListener("keydown", function(event) { if (event.key === "Enter") { getWeather(); } }); async function getWeather() { const city = cityInput.value.trim(); if (!city) { alert("Please enter a city name."); return; } searchButton.disabled = true; searchButton.textContent = "Loading..."; try { // Find the city's coordinates const geoResponse = await fetch( `https://geocoding-api.open-meteo.com/v1/search?name=${encodeURIComponent(city)}&count=1` ); if (!geoResponse.ok) { throw new Error("Location search failed."); } const geoData = await geoResponse.json(); if (!geoData.results || geoData.results.length === 0) { throw new Error("City not found. Try another name."); } const location = geoData.results[0]; // Fetch current weather and 7-day forecast const weatherResponse = await fetch( `https://api.open-meteo.com/v1/forecast?latitude=${location.latitude}&longitude=${location.longitude}¤t=temperature_2m,relative_humidity_2m,precipitation,wind_speed_10m,weather_code&daily=temperature_2m_max,temperature_2m_min,precipitation_sum,weather_code&timezone=auto&forecast_days=7` ); if (!weatherResponse.ok) { throw new Error("Weather data could not be loaded."); } const data = await weatherResponse.json(); document.getElementById("cityName").textContent = `${location.name}, ${location.country}`; document.getElementById("temperature").textContent = `${data.current.temperature_2m}°C`; document.getElementById("humidity").textContent = `${data.current.relative_humidity_2m}%`; document.getElementById("windSpeed").textContent = `${data.current.wind_speed_10m} km/h`; document.getElementById("rainfall").textContent = `${data.current.precipitation} mm`; document.getElementById("weatherCondition").textContent = getWeatherDescription(data.current.weather_code); // Display the 7-day forecast const container = document.getElementById("forecastContainer"); container.replaceChildren(); data.daily.time.forEach((date, index) => { const card = document.createElement("div"); card.className = "forecast-day"; const heading = document.createElement("h3"); heading.textContent = date; const condition = document.createElement("p"); condition.textContent = getWeatherDescription(data.daily.weather_code[index]); const temperature = document.createElement("p"); temperature.textContent = `${data.daily.temperature_2m_min[index]}°C – ${data.daily.temperature_2m_max[index]}°C`; const rain = document.createElement("p"); rain.textContent = `🌧️ ${data.daily.precipitation_sum[index]} mm`; card.append(heading, condition, temperature, rain); container.appendChild(card); }); } catch (error) { alert(error.message); } finally { searchButton.disabled = false; searchButton.textContent = "Check Weather"; } } function getWeatherDescription(code) { if (code === 0) return "☀️ Clear sky"; if ([1, 2, 3].includes(code)) return "⛅ Partly cloudy"; if ([45, 48].includes(code)) return "🌫️ Foggy"; if ([51, 53, 55, 56, 57].includes(code)) return "🌦️ Drizzle"; if ([61, 63, 65, 66, 67, 80, 81, 82].includes(code)) return "🌧️ Rainy"; if ([71, 73, 75, 77, 85, 86].includes(code)) return "❄️ Snowy"; if ([95, 96, 99].includes(code)) return "⛈️ Thunderstorm"; return "🌤️ Weather information"; }
